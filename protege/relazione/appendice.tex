\subsection*{Regole SWRL}
\begin{itemize}\footnotesize
\item \textbf{PublisherGroup}: \texttt{publishedBy(?g, ?c) $\rightarrow$ publishedByGroup(?g, ?c)}
\item \textbf{ParentGroup}: \texttt{publishedByGroup(?g, ?c) $\wedge$ subsidiaryOf(?c, ?p) $\rightarrow$ publishedByGroup(?g, ?p)}
\item \textbf{FirstPartyOnRivalConsole}: \texttt{publishedByGroup(?g, ?h) $\wedge$ availableOn(?g, ?p) $\wedge$ ConsolePlatform(?p) $\wedge$ manufacturedBy(?p, ?m) $\wedge$ manufacturerOf(?h, ?q) $\wedge$ ConsolePlatform(?q) $\wedge$ differentFrom(?h, ?m) $\rightarrow$ FirstPartyOnRivalConsole(?g)}
\item \textbf{SequelByDifferentStudio}: \texttt{sequelOf(?a, ?b) $\wedge$ developedBy(?a, ?x) $\wedge$ developedBy(?b, ?y) $\wedge$ differentFrom(?x, ?y) $\rightarrow$ SequelByDifferentStudio(?a)}
\item \textbf{VeteransReturn}: \texttt{developedBy(?g, ?c) $\wedge$ foundedBy(?c, ?p) $\wedge$ formerEmployeeOf(?p, ?o) $\wedge$ developerOf(?o, ?h) $\wedge$ belongsTo(?g, ?f) $\wedge$ belongsTo(?h, ?f) $\rightarrow$ VeteransReturn(?g)}
\item \textbf{EngineChangeSequel}: \texttt{sequelOf(?a, ?b) $\wedge$ madeWith(?a, ?x) $\wedge$ madeWith(?b, ?y) $\wedge$ differentFrom(?x, ?y) $\rightarrow$ EngineChangeSequel(?a)}
\item \textbf{ExternalEngine}: \texttt{madeWith(?g, ?e) $\wedge$ engineDevelopedBy(?e, ?m) $\wedge$ developedBy(?g, ?d) $\wedge$ differentFrom(?m, ?d) $\rightarrow$ ExternalEngineGame(?g)}
\item \textbf{InHouseEngine}: \texttt{madeWith(?g, ?e) $\wedge$ engineDevelopedBy(?e, ?d) $\wedge$ developedBy(?g, ?d) $\rightarrow$ InHouseEngineGame(?g)}
\item \textbf{ReturningDirector}: \texttt{sequelOf(?a, ?b) $\wedge$ directedBy(?a, ?p) $\wedge$ directedBy(?b, ?p) $\rightarrow$ ReturningDirectorSequel(?a)}
\item \textbf{DefeatedVillain}: \texttt{antagonistOf(?v, ?g) $\wedge$ protagonistOf(?h, ?g) $\wedge$ killedBy(?v, ?h) $\rightarrow$ DefeatedVillain(?v)}
\item \textbf{MustPlay}: \texttt{metacriticScore(?g, ?s) $\wedge$ xsd:integer[>= 95](?s) $\rightarrow$ MustPlayGame(?g)}
\end{itemize}
