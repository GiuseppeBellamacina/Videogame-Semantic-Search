\begin{frame}{Obiettivo del Progetto}
  \begin{block}{Cosa abbiamo costruito}
    Un'\textbf{ontologia OWL 2 di alta qualit\`a} sul dominio dei videogiochi (2010--2026), popolata con dati estratti da Wikidata e interrogabile in linguaggio naturale tramite un'interfaccia web.
  \end{block}

    \begin{itemize}
      \item 78 classi e 57 propriet\`a con semantica DL
      \item 17 classi definite con OWL 2 Restriction
      \item Reasoning a due livelli (OWL 2 RL + SPARQL CONSTRUCT)
      \item Frontend con grafo interattivo e interrogazione NL
    \end{itemize}
    \item \textbf{Casi d'uso}: raccomandazione, analisi di mercato, esplorazione semantica, studio del dominio
  \end{itemize}
\end{frame}

\begin{frame}{Ontologia — Assiomi delle Propriet\`a (Approfondimento)}
  \begin{itemize}\small
    \item \textbf{Inverse sistematiche} (13 coppie): ogni propriet\`a direzionale ha la sua inversa
    \begin{itemize}\scriptsize
      \item \texttt{developedBy} $\leftrightarrow$ \texttt{developerOf}
      \item \texttt{hasCharacter} $\leftrightarrow$ \texttt{appearsIn}
      \item \texttt{belongsTo} $\leftrightarrow$ \texttt{includes}
      \item \texttt{wonAward} $\leftrightarrow$ \texttt{awardWonBy}
      \item \texttt{sequelOf} $\leftrightarrow$ \texttt{hasSequel}
    \end{itemize}
    \item \textbf{Transitive}: \texttt{sequelOf} (catene di sequel), \texttt{partOfFranchise} (sub-franchise $\to$ franchise)
    \item \textbf{Asymmetric + Irreflexive}: la maggior parte delle object property (es.\ \texttt{developedBy} non pu\`o essere simmetrica n\'e riflessiva)
    \item \textbf{3 Property Chain Axiom} (inferenza simmetrica):
    \begin{itemize}\scriptsize
      \item \texttt{sharedFranchiseWith} = \texttt{belongsTo} $\circ$ \texttt{includes}
      \item \texttt{sharesDeveloperWith} = \texttt{developedBy} $\circ$ \texttt{developerOf}
      \item \texttt{sharesPublisherWith} = \texttt{publishedBy} $\circ$ \texttt{publisherOf}
    \end{itemize}
    \item \textbf{owlrl materializza} automaticamente: \texttt{developerOf}, \texttt{platformFor}, \texttt{genreOf}, \texttt{awardWonBy}, \texttt{modeInGame}, \texttt{engineUsedIn}, \texttt{hasSequel}, e le 3 catene simmetriche
  \end{itemize}
\end{frame}

\begin{frame}{Bug Critico — owlrl e DatatypeProperty funzionali}

  \begin{alertblock}{Il problema}
    owlrl inferisce \texttt{owl:sameAs} falsi tra giochi distinti quando \texttt{FunctionalProperty} o \texttt{hasKey} sono applicati a DatatypeProperty: owlrl tratta i valori letterali come individui, causando l'identificazione errata di risorse.
  \end{alertblock}

  \vspace{8pt}

  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Prima del fix}
    \begin{itemize}
      \item 3 giochi classificati \texttt{AwardWinningGame}
      \item 2 sameAs errati tra giochi con stesso Metacritic o data
      \item Classificazioni falsate
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Dopo il fix}
    \begin{itemize}
      \item 1 gioco classificato \texttt{AwardWinningGame} (corretto)
      \item Nessun sameAs spurio
      \item Classificazioni accurate
    \end{itemize}
  \end{columns}

  \vspace{8pt}

  \begin{block}{Soluzione adottata}
    \textbf{Sezione ``DL-only axioms'' separata} nell'ontologia.
    17 assiomi (FunctionalProperty, InverseFunctionalProperty, hasKey su DatatypeProperty) isolati con flag \texttt{--keep-dl-axioms}.
    Funzioni \texttt{strip\_dl\_only\_axioms()} / \texttt{restore\_dl\_only\_axioms()} in \texttt{enrich\_owl.py}.
  \end{block}

\end{frame}
